# 13. GENERIC REACT COMPONENTS

| Component Pattern | Syntax | Description |
| --- | --- | --- |
| Generic Functional Component | `function List<T>({ items, renderItem }: ListProps<T>)` | Allows component to accept any item type while ensuring item callbacks remain strongly typed |
| Generic Arrow Function Component | `const Select = <T extends { id: string }>({ options }: Props<T>) => ...` | Trailing comma `<T,>` prevents JSX parser ambiguity |

**Example:**
```tsx
import React from "react";

// 1. Generic Table / List Props Interface
interface GenericListProps<T> {
  items: T[];
  renderItem: (item: T, index: number) => React.ReactNode;
  keyExtractor: (item: T) => string | number;
  emptyMessage?: string;
}

// 2. Generic Component Declaration
export function GenericList<T>({
  items,
  renderItem,
  keyExtractor,
  emptyMessage = "No items found."
}: GenericListProps<T>): React.JSX.Element {
  if (items.length === 0) {
    return <p className="empty-msg">{emptyMessage}</p>;
  }

  return (
    <ul className="generic-list">
      {items.map((item, idx) => (
        <li key={keyExtractor(item)}>{renderItem(item, idx)}</li>
      ))}
    </ul>
  );
}

// Usage Example:
interface Student {
  id: number;
  name: string;
  grade: string;
}

const students: Student[] = [
  { id: 1, name: "Lucas", grade: "A" },
  { id: 2, name: "Sophia", grade: "A+" }
];

export function Classroom() {
  return (
    <GenericList
      items={students}
      keyExtractor={(student) => student.id}
      renderItem={(student) => (
        <span>
          {student.name} - Grade: {student.grade}
        </span>
      )}
    />
  );
}
```
