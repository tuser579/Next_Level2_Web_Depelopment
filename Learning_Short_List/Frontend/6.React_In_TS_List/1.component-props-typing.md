# 1. TYPING COMPONENT PROPS & RETURN TYPES

| Type / Pattern | Syntax Example | Description |
| --- | --- | --- |
| Props Interface | `interface ButtonProps { label: string; }` | Standard contract for component props |
| Function Component Declaration | `export function Button({ label }: ButtonProps): React.JSX.Element` | Preferred standard function syntax over `React.FC` |
| `React.ReactNode` | `children: React.ReactNode` | Represents anything renderable in JSX (elements, strings, fragments, numbers, null) |
| `React.JSX.Element` | `(): React.JSX.Element` | Exact return type of a JSX element |
| `React.CSSProperties` | `style?: React.CSSProperties` | Type-safe inline CSS style object |

**Example:**
```tsx
import React from "react";

// 1. Props Interface
interface UserCardProps {
  userId: string;
  name: string;
  role: "admin" | "editor" | "viewer";
  avatarUrl?: string; // Optional
  style?: React.CSSProperties; // Typed inline CSS
  children?: React.ReactNode;  // Renderable children
}

// 2. Standard typed Functional Component (Recommended pattern)
export function UserCard({
  userId,
  name,
  role,
  avatarUrl = "https://via.placeholder.com/150",
  style,
  children
}: UserCardProps): React.JSX.Element {
  return (
    <article className="user-card" style={style}>
      <img src={avatarUrl} alt={`${name}'s avatar`} />
      <h3>{name}</h3>
      <span className={`badge badge-${role}`}>{role.toUpperCase()}</span>
      <div className="extra-content">{children}</div>
    </article>
  );
}
```
