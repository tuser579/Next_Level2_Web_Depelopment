# 28. TYPING IN REACT TESTING LIBRARY & VITEST

| Testing Tool | Type Pattern | Description |
| --- | --- | --- |
| Mock Callback | `vi.fn<() => void>()` | Typed mock function for prop testing |
| `RenderResult` | `render(<Component />)` | Typed DOM query container |
| User Events | `userEvent.click(button: HTMLElement)` | Typed user interaction simulator |

**Example:**
```tsx
// Example Vitest / React Testing Library Component Test
import { render, screen } from "@testing-library/react";
import userEvent from "@testing-library/user-event";
import React from "react";
// import { describe, it, expect, vi } from "vitest";

interface CounterButtonProps {
  onIncrement: (count: number) => void;
}

function CounterButton({ onIncrement }: CounterButtonProps) {
  return <button onClick={() => onIncrement(1)}>Add</button>;
}
```
