# 8. USEREF: DOM REFS VS MUTABLE VALUES

| Ref Type | Syntax | Mutable `.current`? |
| --- | --- | --- |
| DOM Element Ref | `useRef<HTMLInputElement>(null)` | Read-only `.current` (React assigns DOM node) |
| Mutable Value Ref | `useRef<number>(0)` or `useRef<number | null>(null)` | Writable `.current` (you can reassign `ref.current = value`) |
| Timer / Interval Ref | `useRef<number | null>(null)` (browser) / `NodeJS.Timeout` (Node) | Holds interval ID across renders |

**Example:**
```tsx
import React, { useRef, useEffect } from "react";

export default function RefDemo() {
  // 1. DOM Ref: Pass initial value `null` + HTMLElement Generic
  // React sets inputRef.current to HTMLInputElement
  const inputRef = useRef<HTMLInputElement>(null);

  // 2. Mutable Value Ref: Does NOT trigger re-renders on mutation
  const renderCountRef = useRef<number>(0);
  const timerIdRef = useRef<number | null>(null);

  useEffect(() => {
    renderCountRef.current += 1;
    inputRef.current?.focus(); // Type-safe DOM access
  });

  const startTimer = () => {
    if (timerIdRef.current !== null) return;
    timerIdRef.current = window.setInterval(() => {
      console.log("Tick");
    }, 1000);
  };

  const stopTimer = () => {
    if (timerIdRef.current !== null) {
      window.clearInterval(timerIdRef.current);
      timerIdRef.current = null;
    }
  };

  return (
    <div>
      <input ref={inputRef} type="text" placeholder="Type here..." />
      <p>Render count: {renderCountRef.current}</p>
      <button onClick={startTimer}>Start</button>
      <button onClick={stopTimer}>Stop</button>
    </div>
  );
}
```
