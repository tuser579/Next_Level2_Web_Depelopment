# 18. TYPING REACT.LAZY & SUSPENSE

| API | Type Definition | Description |
| --- | --- | --- |
| `React.lazy<P>()` | `React.lazy(() => import('./Component'))` | Dynamically loads component returning `{ default: ComponentType<P> }` |
| `<Suspense>` | `fallback: React.ReactNode` | Loading boundary while promise is in flight |

**Example:**
```tsx
import React, { lazy, Suspense } from "react";

// Interface of lazy-loaded component props
interface ChartProps {
  dataPoints: number[];
  colorScheme: "dark" | "light";
}

// Typed dynamic import
const LazyChart = lazy<React.ComponentType<ChartProps>>(
  () => import("./DataChart")
);

export function DashboardView() {
  return (
    <Suspense fallback={<div>Loading Chart Component...</div>}>
      <LazyChart dataPoints={[10, 20, 35, 70]} colorScheme="dark" />
    </Suspense>
  );
}
```
