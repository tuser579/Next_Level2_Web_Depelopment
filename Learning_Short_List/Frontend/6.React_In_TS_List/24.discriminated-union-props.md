# 24. DISCRIMINATED UNION PROPS

| Pattern | Structure | Description |
| --- | --- | --- |
| Discriminated Props Union | `type Props = ButtonVariantProps | LinkVariantProps;` | Forces specific props to exist based on a discriminant tag prop (e.g. `href` required if `variant='link'`) |

**Example:**
```tsx
import React from "react";

// Button variant: requires onClick, forbids href
interface ButtonProps {
  variant: "button";
  onClick: () => void;
  href?: never;
  children: React.ReactNode;
}

// Link variant: requires href, forbids onClick
interface LinkProps {
  variant: "link";
  href: string;
  onClick?: never;
  children: React.ReactNode;
}

type ClickableProps = ButtonProps | LinkProps;

export function ClickableElement(props: ClickableProps) {
  if (props.variant === "link") {
    return <a href={props.href} className="btn-link">{props.children}</a>;
  }

  return <button onClick={props.onClick} className="btn">{props.children}</button>;
}

// ✅ Valid:
// <ClickableElement variant="button" onClick={() => {}}>Submit</ClickableElement>
// <ClickableElement variant="link" href="/home">Go Home</ClickableElement>
// ❌ Invalid (Compile Error):
// <ClickableElement variant="button" href="/home">Submit</ClickableElement>
```
