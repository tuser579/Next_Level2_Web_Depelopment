# 15. TYPING FORMS & CONTROLLED INPUTS

| Element Type | Event Generic | Value Extraction |
| --- | --- | --- |
| `<input type="text">` | `React.ChangeEvent<HTMLInputElement>` | `e.target.value` |
| `<input type="checkbox">` | `React.ChangeEvent<HTMLInputElement>` | `e.target.checked` |
| `<select>` | `React.ChangeEvent<HTMLSelectElement>` | `e.target.value` |
| `<textarea>` | `React.ChangeEvent<HTMLTextAreaElement>` | `e.target.value` |

**Example:**
```tsx
import React, { useState } from "react";

interface SignupFormData {
  username: string;
  email: string;
  plan: "free" | "pro" | "enterprise";
  subscribed: boolean;
}

export function SignupForm() {
  const [formData, setFormData] = useState<SignupFormData>({
    username: "",
    email: "",
    plan: "free",
    subscribed: false
  });

  const handleChange = (
    e: React.ChangeEvent<HTMLInputElement | HTMLSelectElement | HTMLTextAreaElement>
  ) => {
    const target = e.target;
    const value = target.type === "checkbox" 
      ? (target as HTMLInputElement).checked 
      : target.value;

    setFormData((prev) => ({
      ...prev,
      [target.name]: value
    }));
  };

  const handleSubmit = (e: React.FormEvent<HTMLFormElement>) => {
    e.preventDefault();
    console.log("Submitted payload:", formData);
  };

  return (
    <form onSubmit={handleSubmit}>
      <input name="username" value={formData.username} onChange={handleChange} />
      <input name="email" type="email" value={formData.email} onChange={handleChange} />
      <select name="plan" value={formData.plan} onChange={handleChange}>
        <option value="free">Free</option>
        <option value="pro">Pro</option>
        <option value="enterprise">Enterprise</option>
      </select>
      <input
        name="subscribed"
        type="checkbox"
        checked={formData.subscribed}
        onChange={handleChange}
      />
      <button type="submit">Create Account</button>
    </form>
  );
}
```
