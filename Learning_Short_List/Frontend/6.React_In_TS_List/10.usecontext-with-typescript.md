# 10. TYPING CONTEXT API & CUSTOM HOOKS

| Step | Pattern | Description |
| --- | --- | --- |
| Context Value Interface | `interface AuthContextType` | Full contract of state variables and methods exposed |
| `createContext` Initialization | `createContext<AuthContextType | null>(null)` | Initialize with `null` to avoid fake mock defaults |
| Custom Hook with Guard | `useAuth()` checking `!context` | Throws error if consumed outside `<Provider>`, narrowing type away from null |

**Example:**
```tsx
import React, { createContext, useContext, useState } from "react";

// 1. Define Context State & Actions Interface
interface AuthUser {
  id: string;
  name: string;
  email: string;
}

interface AuthContextType {
  user: AuthUser | null;
  isAuthenticated: boolean;
  login: (credentials: { email: string }) => Promise<void>;
  logout: () => void;
}

// 2. Create Context (Initialized with null)
const AuthContext = createContext<AuthContextType | null>(null);

// 3. Typed Provider Component
export function AuthProvider({ children }: { children: React.ReactNode }) {
  const [user, setUser] = useState<AuthUser | null>(null);

  const login = async ({ email }: { email: string }) => {
    setUser({ id: "usr_101", name: "Alex", email });
  };

  const logout = () => setUser(null);

  return (
    <AuthContext.Provider value={{ user, isAuthenticated: !!user, login, logout }}>
      {children}
    </AuthContext.Provider>
  );
}

// 4. Custom Hook with Non-Null Guard (Narrows away null!)
export function useAuth(): AuthContextType {
  const context = useContext(AuthContext);
  if (!context) {
    throw new Error("useAuth must be consumed within an <AuthProvider>");
  }
  return context; // Guaranteed to be strictly AuthContextType (no null!)
}
```
