# 2. OPTIONAL PROPS & DEFAULT VALUES

| Pattern | Syntax Example | Description |
| --- | --- | --- |
| Optional Modifier (`?`) | `variant?: 'primary' | 'secondary'` | Property may be passed or omitted (`undefined`) |
| Default Destructuring | `({ variant = 'primary' }: Props)` | Supplies fallback default value when prop is undefined |
| Union Prop Types | `size: 'sm' | 'md' | 'lg'` | Constrains prop to specific allowed literal values |

**Example:**
```tsx
import React from "react";

type ButtonVariant = "primary" | "secondary" | "danger" | "ghost";
type ButtonSize = "sm" | "md" | "lg";

interface ActionButtonProps {
  label: string;
  variant?: ButtonVariant; // Optional with union
  size?: ButtonSize;       // Optional with union
  isLoading?: boolean;     // Optional boolean
  onClick?: () => void;    // Optional callback
}

export function ActionButton({
  label,
  variant = "primary", // Default value
  size = "md",         // Default value
  isLoading = false,   // Default value
  onClick
}: ActionButtonProps) {
  return (
    <button
      className={`btn btn-${variant} btn-${size}`}
      disabled={isLoading}
      onClick={onClick}
    >
      {isLoading ? "Loading..." : label}
    </button>
  );
}
```
