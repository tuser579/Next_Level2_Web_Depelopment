# 23. GLOBAL STATE: TYPING ZUSTAND & REDUX TOOLKIT

| Library | Pattern | Description |
| --- | --- | --- |
| Zustand | `create<StoreState>()((set, get) => ...)` | Complete type-safety for actions and state without extra boilerplate |
| Redux Toolkit | `createSlice({ ... })`, `RootState`, `AppDispatch` | Typed reducers with `PayloadAction<T>` |

**Example:**
```tsx
// === 1. Typed Zustand Store ===
import { create } from "zustand";

interface ThemeStore {
  theme: "light" | "dark";
  accentColor: string;
  setTheme: (theme: "light" | "dark") => void;
  setAccentColor: (color: string) => void;
}

export const useThemeStore = create<ThemeStore>()((set) => ({
  theme: "light",
  accentColor: "#3b82f6",
  setTheme: (theme) => set({ theme }),
  setAccentColor: (accentColor) => set({ accentColor })
}));

// === 2. Redux Toolkit Slice Example ===
import { createSlice, PayloadAction } from "@reduxjs/toolkit";

interface CounterState {
  value: number;
}

const initialState: CounterState = { value: 0 };

export const counterSlice = createSlice({
  name: "counter",
  initialState,
  reducers: {
    incrementByAmount: (state, action: PayloadAction<number>) => {
      state.value += action.payload;
    }
  }
});
```
