# 6. TYPING THE USESTATE HOOK

| State Category | Syntax Example | Description |
| --- | --- | --- |
| Inferred Primitive | `const [count, setCount] = useState(0);` | TypeScript infers `number` automatically |
| Explicit Generic State | `const [user, setUser] = useState<User | null>(null);` | Explicit union type for asynchronous data starting as null |
| Array of Objects State | `const [items, setItems] = useState<Item[]>([]);` | Avoids `never[]` inferred array type |
| Functional State Updater | `setUser(prev => prev ? { ...prev, name: 'New' } : null)` | Parameter `prev` is strictly typed |

**Example:**
```tsx
import React, { useState } from "react";

interface UserProfile {
  id: string;
  username: string;
  email: string;
  roles: string[];
}

export default function StateDemo() {
  // 1. Primitive: Inferred as number
  const [likes, setLikes] = useState(0);

  // 2. Complex Object Union: Explicitly UserProfile | null
  const [user, setUser] = useState<UserProfile | null>(null);

  // 3. Array State: Must provide generic <T[]> to avoid never[]
  const [tags, setTags] = useState<string[]>([]);

  const handleAddTag = (newTag: string) => {
    setTags((prevTags) => [...prevTags, newTag]); // prevTags is strictly string[]
  };

  const handleUpdateUsername = (newName: string) => {
    setUser((prevUser) => {
      if (!prevUser) return null;
      return { ...prevUser, username: newName };
    });
  };

  return (
    <div>
      <button onClick={() => setLikes((c) => c + 1)}>Likes: {likes}</button>
      <button onClick={() => handleAddTag("React")}>Add Tag</button>
    </div>
  );
}
```
