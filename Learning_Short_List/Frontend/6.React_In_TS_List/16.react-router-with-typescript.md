# 16. REACT ROUTER & TYPESCRIPT

| Hook / Utility | Generic Syntax | Description |
| --- | --- | --- |
| `useParams<Params>()` | `useParams<{ id: string }>()` | Strongly types extracted URL parameters |
| `useLocation<State>()` | `useLocation<{ from: string }>()` | Types navigation location state |
| `LoaderFunctionArgs` | `loader: async ({ params }: LoaderFunctionArgs)` | Types React Router loader arguments |

**Example:**
```tsx
import React from "react";
import { useParams, useNavigate, useLocation } from "react-router-dom";

type RouteParams = {
  organizationId: string;
  projectId: string;
};

interface LocationState {
  fromDashboard?: boolean;
}

export function ProjectDetails() {
  // 1. Strongly typed route params
  const { organizationId, projectId } = useParams<RouteParams>();
  const navigate = useNavigate();
  
  // 2. Typed location state
  const location = useLocation();
  const state = location.state as LocationState | null;

  return (
    <div>
      <h2>Project: {projectId}</h2>
      <p>Org: {organizationId}</p>
      {state?.fromDashboard && <p>Navigated from Dashboard</p>}
      <button onClick={() => navigate("/projects")}>Back to Projects</button>
    </div>
  );
}
```
