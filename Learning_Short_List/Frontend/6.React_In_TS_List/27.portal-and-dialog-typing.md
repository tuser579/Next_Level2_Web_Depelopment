# 27. TYPING REACT PORTALS & DIALOGS

| API / Element | Type | Description |
| --- | --- | --- |
| `createPortal()` | `createPortal(children: ReactNode, container: Element): ReactPortal` | Teleports JSX to target DOM node |
| `<dialog>` element | `useRef<HTMLDialogElement>(null)` | Accesses `.showModal()` and `.close()` methods |

**Example:**
```tsx
import React, { useRef } from "react";
import { createPortal } from "react-dom";

interface PortalDrawerProps {
  isOpen: boolean;
  onClose: () => void;
  children: React.ReactNode;
}

export function PortalDrawer({ isOpen, onClose, children }: PortalDrawerProps) {
  if (!isOpen) return null;

  return createPortal(
    <div className="drawer-overlay" onClick={onClose}>
      <aside className="drawer-panel" onClick={(e) => e.stopPropagation()}>
        <button onClick={onClose}>Close</button>
        {children}
      </aside>
    </div>,
    document.body
  );
}

export function NativeDialog() {
  const dialogRef = useRef<HTMLDialogElement>(null);

  return (
    <>
      <button onClick={() => dialogRef.current?.showModal()}>Open Dialog</button>
      <dialog ref={dialogRef}>
        <p>Native Dialog Window</p>
        <button onClick={() => dialogRef.current?.close()}>Close</button>
      </dialog>
    </>
  );
}
```
