# 22. TANSTACK QUERY (REACT QUERY) & TYPESCRIPT

| Hook | Generics | Description |
| --- | --- | --- |
| `useQuery<TData, TError>()` | `useQuery<User[], Error>` | Queries data with typed response and error |
| `useMutation<TData, TError, TVariables>()` | `useMutation<Todo, Error, NewTodoInput>` | Mutates data with typed payload variables |

**Example:**
```tsx
import { useQuery, useMutation, useQueryClient } from "@tanstack/react-query";

interface User {
  id: number;
  name: string;
  email: string;
}

interface CreateUserInput {
  name: string;
  email: string;
}

// 1. Typed Query Hook
export function useUsers() {
  return useQuery<User[], Error>({
    queryKey: ["users"],
    queryFn: async (): Promise<User[]> => {
      const res = await fetch("https://jsonplaceholder.typicode.com/users");
      if (!res.ok) throw new Error("Failed to fetch users");
      return res.json() as Promise<User[]>;
    }
  });
}

// 2. Typed Mutation Hook
export function useCreateUser() {
  const queryClient = useQueryClient();

  return useMutation<User, Error, CreateUserInput>({
    mutationFn: async (newUser: CreateUserInput): Promise<User> => {
      const res = await fetch("/api/users", {
        method: "POST",
        body: JSON.stringify(newUser)
      });
      return res.json() as Promise<User>;
    },
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ["users"] });
    }
  });
}
```
