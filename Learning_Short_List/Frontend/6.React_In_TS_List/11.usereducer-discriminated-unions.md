# 11. TYPING USEREDUCER WITH DISCRIMINATED UNIONS

| Element | Pattern | Description |
| --- | --- | --- |
| State Interface | `interface State` | Structure of the reducer state tree |
| Action Discriminated Union | `type Action = { type: 'ADD' } | { type: 'RESET' }` | Exhaustive set of all possible action types and payloads |
| Reducer Function | `function reducer(state: State, action: Action): State` | Pure state transformation function |

**Example:**
```tsx
import React, { useReducer } from "react";

// 1. State Type
interface CartItem {
  id: string;
  title: string;
  quantity: number;
}

interface CartState {
  items: CartItem[];
  totalQuantity: number;
}

// 2. Discriminated Union for Actions
type CartAction =
  | { type: "ADD_ITEM"; payload: CartItem }
  | { type: "REMOVE_ITEM"; payload: { id: string } }
  | { type: "UPDATE_QTY"; payload: { id: string; quantity: number } }
  | { type: "CLEAR_CART" };

// 3. Reducer with Exhaustive Type Safety
function cartReducer(state: CartState, action: CartAction): CartState {
  switch (action.type) {
    case "ADD_ITEM":
      return {
        ...state,
        items: [...state.items, action.payload],
        totalQuantity: state.totalQuantity + action.payload.quantity
      };
    case "REMOVE_ITEM":
      return {
        ...state,
        items: state.items.filter((item) => item.id !== action.payload.id)
      };
    case "UPDATE_QTY":
      return {
        ...state,
        items: state.items.map((item) =>
          item.id === action.payload.id
            ? { ...item, quantity: action.payload.quantity }
            : item
        )
      };
    case "CLEAR_CART":
      return { items: [], totalQuantity: 0 };
  }
}

export function CartComponent() {
  const [state, dispatch] = useReducer(cartReducer, { items: [], totalQuantity: 0 });

  return (
    <button
      onClick={() =>
        dispatch({
          type: "ADD_ITEM",
          payload: { id: "p1", title: "Book", quantity: 1 }
        })
      }
    >
      Add Item
    </button>
  );
}
```
