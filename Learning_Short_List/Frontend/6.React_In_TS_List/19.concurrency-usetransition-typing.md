# 19. TYPING USETRANSITION & USEDEFERREDVALUE

| Hook | Signature | Description |
| --- | --- | --- |
| `useTransition()` | `[isPending: boolean, startTransition: (callback: () => void) => void]` | Defines non-blocking state updates |
| `useDeferredValue<T>(value: T)` | `useDeferredValue<T>(value: T): T` | Defers updating heavy calculation |

**Example:**
```tsx
import React, { useState, useTransition, useDeferredValue } from "react";

export function SearchFilter() {
  const [query, setQuery] = useState<string>("");
  const [filteredQuery, setFilteredQuery] = useState<string>("");
  const [isPending, startTransition] = useTransition();

  const handleSearch = (e: React.ChangeEvent<HTMLInputElement>) => {
    setQuery(e.target.value); // Urgent

    startTransition(() => {
      setFilteredQuery(e.target.value); // Non-urgent transition
    });
  };

  const deferredValue = useDeferredValue<string>(query);

  return (
    <div>
      <input type="text" value={query} onChange={handleSearch} />
      {isPending && <p>Filtering list in background...</p>}
      <p>Deferred Value: {deferredValue}</p>
    </div>
  );
}
```
