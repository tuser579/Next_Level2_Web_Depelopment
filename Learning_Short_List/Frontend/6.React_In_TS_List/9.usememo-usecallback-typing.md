# 9. TYPING USEMEMO & USECALLBACK

| Hook | Type Parameter | Inferred Type |
| --- | --- | --- |
| `useMemo<T>()` | Explicit or inferred result type | Returns memoized `T` value |
| `useCallback<T>()` | Function signature type | Returns memoized function signature `(...args: Args) => Return` |

**Example:**
```tsx
import React, { useState, useMemo, useCallback } from "react";

interface Product {
  id: number;
  name: string;
  price: number;
}

export default function MemoDemo({ products }: { products: Product[] }) {
  const [minPrice, setMinPrice] = useState<number>(50);

  // 1. useMemo with explicit / inferred return type
  const filteredProducts = useMemo<Product[]>(() => {
    return products.filter((p) => p.price >= minPrice);
  }, [products, minPrice]);

  // 2. useCallback with typed parameters
  const handleProductSelect = useCallback((productId: number): void => {
    console.log(`Product chosen: ${productId}`);
  }, []);

  return (
    <div>
      <input
        type="number"
        value={minPrice}
        onChange={(e: React.ChangeEvent<HTMLInputElement>) =>
          setMinPrice(Number(e.target.value))
        }
      />
      <ul>
        {filteredProducts.map((p) => (
          <li key={p.id} onClick={() => handleProductSelect(p.id)}>
            {p.name} - ${p.price}
          </li>
        ))}
      </ul>
    </div>
  );
}
```
