# 20. REACT 19: TYPING ACTIONS & USEACTIONSTATE

| Hook / Feature | Type Signature | Description |
| --- | --- | --- |
| Action Function | `type ActionFn<S, P> = (state: S, payload: P) => Promise<S>` | Typed async action handler |
| `useActionState<S, P>()` | `[state: S, formAction: (payload: P) => void, isPending: boolean]` | Manages action state and execution status |
| `useOptimistic<S, U>()` | `[optimisticState: S, setOptimistic: (update: U) => void]` | Types optimistic state transitions |

**Example:**
```tsx
import React, { useActionState, useOptimistic } from "react";

interface FormState {
  message: string | null;
  error: string | null;
  success: boolean;
}

// Typed Async Form Action
async function submitFeedbackAction(
  prevState: FormState,
  formData: FormData
): Promise<FormState> {
  const feedback = formData.get("feedback") as string;

  if (!feedback || feedback.length < 5) {
    return { message: null, error: "Feedback must be at least 5 chars", success: false };
  }

  // Simulate API call
  await new Promise((resolve) => setTimeout(resolve, 800));

  return { message: "Feedback submitted successfully!", error: null, success: true };
}

export function FeedbackForm() {
  const [state, formAction, isPending] = useActionState<FormState, FormData>(
    submitFeedbackAction,
    { message: null, error: null, success: false }
  );

  return (
    <form action={formAction}>
      <textarea name="feedback" placeholder="Your feedback..." required />
      <button type="submit" disabled={isPending}>
        {isPending ? "Submitting..." : "Send Feedback"}
      </button>
      {state.error && <p style={{ color: "red" }}>{state.error}</p>}
      {state.message && <p style={{ color: "green" }}>{state.message}</p>}
    </form>
  );
}
```
