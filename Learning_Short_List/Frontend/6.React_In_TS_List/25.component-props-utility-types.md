# 25. REACT PROPS UTILITY TYPES

| Utility Type | Input | Description |
| --- | --- | --- |
| `React.ComponentProps<T>` | `typeof CustomButton` | Extracts the complete Props type of a component |
| `React.ComponentPropsWithoutRef<T>` | `'button'` / `typeof CustomButton` | Extracts props omitting `ref` |
| `React.ElementRef<T>` | `typeof CustomInput` | Extracts the underlying Ref DOM element type |

**Example:**
```tsx
import React from "react";

// Base component
function PrimaryButton({ label, priority }: { label: string; priority: number }) {
  return <button>{label} ({priority})</button>;
}

// 1. Extracting props of an existing component
type PrimaryButtonProps = React.ComponentProps<typeof PrimaryButton>;
// { label: string; priority: number; }

// 2. Wrapping a third-party button while omitting conflicting props
type SuperButtonProps = Omit<React.ComponentPropsWithoutRef<"button">, "color"> & {
  customColor: string;
};

export function SuperButton({ customColor, ...rest }: SuperButtonProps) {
  return <button style={{ backgroundColor: customColor }} {...rest} />;
}
```
