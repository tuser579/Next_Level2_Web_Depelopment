# 3. TYPING CHILDREN & RENDER PROPS

| Children Type | Syntax Example | Description |
| --- | --- | --- |
| `React.ReactNode` | `children: React.ReactNode` | Most versatile; accepts JSX elements, strings, numbers, booleans, null, fragments |
| `React.ReactElement` | `icon: React.ReactElement` | Restricts slot to strictly a single valid JSX element (disallows primitives) |
| Render Props Function | `children: (data: T) => React.ReactNode` | Function that receives state data and returns renderable JSX |
| Named Slot Props | `headerSlot?: React.ReactNode` | Multiple distinct named slot placeholders on a component |

**Example:**
```tsx
import React from "react";

// 1. Named Slot Props + Standard Children
interface ModalProps {
  isOpen: boolean;
  onClose: () => void;
  headerSlot?: React.ReactNode;
  footerSlot?: React.ReactNode;
  icon?: React.ReactElement; // Strictly a single JSX element
  children: React.ReactNode;
}

export function ModalDialog({
  isOpen,
  onClose,
  headerSlot,
  footerSlot,
  icon,
  children
}: ModalProps) {
  if (!isOpen) return null;

  return (
    <div className="modal-overlay" onClick={onClose}>
      <div className="modal-card" onClick={(e) => e.stopPropagation()}>
        {icon && <div className="modal-icon">{icon}</div>}
        {headerSlot && <header className="modal-header">{headerSlot}</header>}
        <div className="modal-body">{children}</div>
        {footerSlot && <footer className="modal-footer">{footerSlot}</footer>}
      </div>
    </div>
  );
}

// 2. Render Props Pattern
interface DataFetcherProps<T> {
  fetchUrl: string;
  children: (data: T | null, isLoading: boolean) => React.ReactNode;
}
```
