# 14. TYPING FORWARDREF & USEIMPERATIVEHANDLE

| Generic Parameter | Position | Description |
| --- | --- | --- |
| Ref Target Type (`T`) | First Generic (`forwardRef<T, P>`) | The DOM Element type (e.g. `HTMLInputElement`) or custom handle interface |
| Props Type (`P`) | Second Generic (`forwardRef<T, P>`) | The component props interface |

**Example:**
```tsx
import React, { forwardRef, useImperativeHandle, useRef } from "react";

// === 1. Standard forwardRef to Native Input ===
interface TextInputProps {
  label: string;
  error?: string;
}

export const CustomTextInput = forwardRef<HTMLInputElement, TextInputProps>(
  ({ label, error }, ref) => {
    return (
      <div className="input-group">
        <label>{label}</label>
        <input ref={ref} type="text" className={error ? "input-error" : ""} />
        {error && <span className="error-text">{error}</span>}
      </div>
    );
  }
);

CustomTextInput.displayName = "CustomTextInput";

// === 2. forwardRef with useImperativeHandle (Exposing Custom API) ===
export interface VideoPlayerHandle {
  playVideo: () => void;
  pauseVideo: () => void;
  seekTo: (seconds: number) => void;
}

export const VideoPlayer = forwardRef<VideoPlayerHandle, { src: string }>(
  ({ src }, ref) => {
    const videoRef = useRef<HTMLVideoElement>(null);

    useImperativeHandle(ref, () => ({
      playVideo: () => videoRef.current?.play(),
      pauseVideo: () => videoRef.current?.pause(),
      seekTo: (seconds: number) => {
        if (videoRef.current) videoRef.current.currentTime = seconds;
      }
    }));

    return <video ref={videoRef} src={src} />;
  }
);
```
