# 5. TYPING REACT EVENTS & HANDLERS

| Event Type | Target Element Generic | Trigger |
| --- | --- | --- |
| `React.MouseEvent<T>` | `React.MouseEvent<HTMLButtonElement>` | Mouse clicks, hover, mouse down/up |
| `React.ChangeEvent<T>` | `React.ChangeEvent<HTMLInputElement>` | Text typing, checkbox toggle, select dropdown change |
| `React.FormEvent<T>` | `React.FormEvent<HTMLFormElement>` | Form submission (`onSubmit`) |
| `React.KeyboardEvent<T>` | `React.KeyboardEvent<HTMLInputElement>` | Key down/up events (`e.key === 'Enter'`) |
| `React.FocusEvent<T>` | `React.FocusEvent<HTMLInputElement>` | Focus and Blur events on inputs |
| `React.DragEvent<T>` | `React.DragEvent<HTMLDivElement>` | Drag and drop interactions |

**Example:**
```tsx
import React, { useState } from "react";

export default function EventTypingDemo() {
  const [searchTerm, setSearchTerm] = useState("");

  // 1. Typed Change Event on Input Element
  const handleInputChange = (e: React.ChangeEvent<HTMLInputElement>) => {
    setSearchTerm(e.target.value);
  };

  // 2. Typed Form Submit Event
  const handleFormSubmit = (e: React.FormEvent<HTMLFormElement>) => {
    e.preventDefault();
    console.log("Searching for:", searchTerm);
  };

  // 3. Typed Button Mouse Click Event
  const handleButtonClick = (e: React.MouseEvent<HTMLButtonElement>) => {
    console.log("Button clicked at coordinates:", e.clientX, e.clientY);
  };

  // 4. Typed Keyboard Key Event
  const handleKeyDown = (e: React.KeyboardEvent<HTMLInputElement>) => {
    if (e.key === "Enter") {
      console.log("Enter key pressed!");
    }
  };

  return (
    <form onSubmit={handleFormSubmit}>
      <input
        type="text"
        value={searchTerm}
        onChange={handleInputChange}
        onKeyDown={handleKeyDown}
        placeholder="Type query..."
      />
      <button type="button" onClick={handleButtonClick}>
        Clear
      </button>
      <button type="submit">Submit</button>
    </form>
  );
}
```
