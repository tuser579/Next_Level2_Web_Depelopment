# 30. ADVANCED REACT & TS ARCHITECTURAL PATTERNS

| Pattern | Architecture | Description |
| --- | --- | --- |
| Compound Components | `Tabs.List`, `Tabs.Tab`, `Tabs.Panel` | Shared implicit context between subcomponents with strong typing |
| Controlled / Uncontrolled Polymorphism | `useControllableState` | Supports both controlled `value` and uncontrolled `defaultValue` |
| As Prop Polymorphic Pattern | `<Button as='a' href='...'>` | Type-safe dynamic HTML element switching |

**Example:**
```tsx
import React, { createContext, useContext, useState } from "react";

// === 1. Type-Safe Compound Components (Tabs Pattern) ===
interface TabsContextType {
  activeTab: string;
  setActiveTab: (id: string) => void;
}

const TabsContext = createContext<TabsContextType | null>(null);

function useTabsContext(): TabsContextType {
  const ctx = useContext(TabsContext);
  if (!ctx) throw new Error("Tabs subcomponents must be inside <Tabs>");
  return ctx;
}

export function Tabs({
  defaultTab,
  children
}: {
  defaultTab: string;
  children: React.ReactNode;
}) {
  const [activeTab, setActiveTab] = useState<string>(defaultTab);

  return (
    <TabsContext.Provider value={{ activeTab, setActiveTab }}>
      <div className="tabs-container">{children}</div>
    </TabsContext.Provider>
  );
}

Tabs.Tab = function Tab({ id, children }: { id: string; children: React.ReactNode }) {
  const { activeTab, setActiveTab } = useTabsContext();
  return (
    <button
      className={`tab-btn ${activeTab === id ? "active" : ""}`}
      onClick={() => setActiveTab(id)}
    >
      {children}
    </button>
  );
};

Tabs.Panel = function Panel({ id, children }: { id: string; children: React.ReactNode }) {
  const { activeTab } = useTabsContext();
  if (activeTab !== id) return null;
  return <div className="tab-panel">{children}</div>;
};

// Usage:
// <Tabs defaultTab="profile">
//   <Tabs.Tab id="profile">Profile</Tabs.Tab>
//   <Tabs.Tab id="settings">Settings</Tabs.Tab>
//   <Tabs.Panel id="profile">Profile Content</Tabs.Panel>
//   <Tabs.Panel id="settings">Settings Content</Tabs.Panel>
// </Tabs>
```
