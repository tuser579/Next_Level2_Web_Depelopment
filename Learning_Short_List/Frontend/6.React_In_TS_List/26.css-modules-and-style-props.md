# 26. TYPING STYLES, CSS MODULES & CLASSNAMES

| Approach | Type | Description |
| --- | --- | --- |
| Inline `style` Prop | `React.CSSProperties` | Type-safe camelCase CSS properties |
| CSS Modules Ambient Types | `declare module '*.module.css'` | Generates type mapping for class names |
| Tailwind / Clsx Class Names | `string | (string | boolean | undefined)[]` | Conditional class compositions |

**Example:**
```tsx
import React from "react";

interface BadgeProps {
  text: string;
  colorScheme?: "success" | "warning" | "danger";
  style?: React.CSSProperties; // Typed inline styles
}

export function Badge({ text, colorScheme = "success", style }: BadgeProps) {
  const baseStyle: React.CSSProperties = {
    padding: "4px 8px",
    borderRadius: "6px",
    display: "inline-block",
    fontSize: "0.75rem",
    fontWeight: 600,
    ...style // Merge with caller styles
  };

  return (
    <span className={`badge badge-${colorScheme}`} style={baseStyle}>
      {text}
    </span>
  );
}
```
