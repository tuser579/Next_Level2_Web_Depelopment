# 21. REACT 19: TYPING THE 'USE' API

| `use()` Target | Signature | Description |
| --- | --- | --- |
| Promise Unwrapping | `use<T>(promise: Promise<T>): T` | Suspends component until Promise resolves to type `T` |
| Context Unwrapping | `use<T>(context: React.Context<T>): T` | Conditionally unwraps React context values |

**Example:**
```tsx
import React, { use, Suspense } from "react";

interface ArticleData {
  id: string;
  title: string;
  content: string;
}

function ArticleContent({ articlePromise }: { articlePromise: Promise<ArticleData> }) {
  // use() directly unwraps Promise<ArticleData> into ArticleData
  const article = use<ArticleData>(articlePromise);

  return (
    <article>
      <h2>{article.title}</h2>
      <p>{article.content}</p>
    </article>
  );
}

export function ArticleViewer({ promise }: { promise: Promise<ArticleData> }) {
  return (
    <Suspense fallback={<div>Loading Article...</div>}>
      <ArticleContent articlePromise={promise} />
    </Suspense>
  );
}
```
