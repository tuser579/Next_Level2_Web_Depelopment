# 25. BROWSER STORAGE & WEB APIS

| API / Storage | Capacity / Scope | Description |
| --- | --- | --- |
| `localStorage` | ~5-10MB (Persistent) | Stores key-value strings; survives browser restarts |
| `sessionStorage` | ~5MB (Tab Session) | Stores key-value strings; cleared when tab is closed |
| `navigator.clipboard` | Async API | Reads and writes text/images to system clipboard (`writeText()`) |
| `IntersectionObserver` | Observer API | Detects when elements enter/exit viewport (Infinite scroll & Lazy loading) |
| `ResizeObserver` | Observer API | Monitors element dimension changes efficiently |

**Example:**
```javascript
// 1. JSON Storage Utility for localStorage
const storage = {
  set(key, val) {
    localStorage.setItem(key, JSON.stringify(val));
  },
  get(key, fallback = null) {
    const data = localStorage.getItem(key);
    return data ? JSON.parse(data) : fallback;
  },
  remove(key) {
    localStorage.removeItem(key);
  }
};

storage.set("preferences", { theme: "dark", fontSize: 16 });
console.log(storage.get("preferences").theme); // "dark"

// 2. IntersectionObserver (Lazy Loading Images & Animations)
const imageObserver = new IntersectionObserver((entries, observer) => {
  entries.forEach((entry) => {
    if (entry.isIntersecting) {
      const img = entry.target;
      img.src = img.dataset.src; // Swap placeholder with real image
      img.classList.add("fade-in");
      observer.unobserve(img); // Stop observing once loaded
    }
  });
}, { threshold: 0.1 });

document.querySelectorAll("img[data-src]").forEach(img => imageObserver.observe(img));

// 3. Clipboard API
async function copyToClipboard(text) {
  await navigator.clipboard.writeText(text);
  console.log("Copied to clipboard!");
}
```
