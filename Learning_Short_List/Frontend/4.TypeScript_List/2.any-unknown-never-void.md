# 2. SPECIAL TYPES: ANY, UNKNOWN, NEVER & VOID

| Type | Safety Level | Description |
| --- | --- | --- |
| `any` | 🚨 Unsafe | Disables all type checking; bypasses TypeScript safety (avoid when possible) |
| `unknown` | 🛡️ Type-Safe | Can hold any value, but operations require type narrowing / verification first |
| `void` | Function Return | Indicates function returns no value (`return;` or `undefined`) |
| `never` | Unreachable / Empty | Represents values that NEVER occur (e.g. infinite loops, functions throwing errors, exhaustive checks) |

**Example:**
```typescript
// 1. any (Unsafe - bypasses compiler)
let dataAny: any = "hello";
dataAny.nonExistentMethod(); // No compiler error, crashes at runtime! 💥

// 2. unknown (Safe alternative to any)
let dataUnknown: unknown = "https://api.example.com";

// Type narrowing required before performing actions!
if (typeof dataUnknown === "string") {
  console.log(dataUnknown.toUpperCase()); // ✅ Safe!
}

// 3. void (Functions that return nothing)
function logNotification(message: string): void {
  console.log(`[LOG]: ${message}`);
}

// 4. never (Unreachable / Error throwing / Exhaustive Checks)
function throwError(errorMessage: string): never {
  throw new Error(errorMessage);
}

type TrafficLight = "red" | "yellow" | "green";

function handleTrafficLight(light: TrafficLight): string {
  switch (light) {
    case "red": return "Stop";
    case "yellow": return "Caution";
    case "green": return "Go";
    default: {
      // Exhaustive checking: TypeScript ensures all cases were covered
      const _exhaustiveCheck: never = light;
      return _exhaustiveCheck;
    }
  }
}
```
