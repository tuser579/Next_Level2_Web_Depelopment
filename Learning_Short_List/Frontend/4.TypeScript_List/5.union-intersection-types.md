# 5. UNION, INTERSECTION & DISCRIMINATED UNIONS

| Pattern | Syntax | Description |
| --- | --- | --- |
| Union Types | `A | B` | Value can be OF TYPE `A` OR TYPE `B` |
| Intersection Types | `A & B` | Combines multiple types into ONE containing all combined properties |
| Discriminated Union | `type T = A | B` (shared literal tag) | Unions containing a common literal property (e.g. `kind` / `status`) for bulletproof type narrowing |
| Pattern Matching | `switch (action.type)` | Compiler narrows exact branch shape automatically |

**Example:**
```typescript
// 1. Discriminated Union Pattern (Best practice for State & Reducers)
interface LoadingState {
  status: "loading"; // Discriminant tag
}

interface SuccessState {
  status: "success"; // Discriminant tag
  data: string[];
}

interface ErrorState {
  status: "error";   // Discriminant tag
  error: Error;
}

type NetworkState = LoadingState | SuccessState | ErrorState;

// 2. Exhaustive Narrowing via Discriminant tag
function renderState(state: NetworkState) {
  switch (state.status) {
    case "loading":
      return "⏳ Loading data, please wait...";
    case "success":
      return `✅ Loaded items: ${state.data.join(", ")}`;
    case "error":
      return `❌ Error occurred: ${state.error.message}`;
  }
}

// 3. Intersection Types (&)
interface Loggable {
  logId: string;
}
interface Serializable {
  serialize(): string;
}

type Entity = Loggable & Serializable;

const auditItem: Entity = {
  logId: "audit_99",
  serialize: () => JSON.stringify({ id: "audit_99" })
};
```
