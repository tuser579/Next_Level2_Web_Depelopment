# 10. TYPE NARROWING & BUILT-IN GUARDS

| Guard Mechanism | Condition | Description |
| --- | --- | --- |
| `typeof` guard | `typeof x === 'string'` | Narrows primitive types (`'string'`, `'number'`, `'boolean'`, `'symbol'`, `'bigint'`, `'function'`) |
| `instanceof` guard | `x instanceof Date` | Narrows class / constructor prototype instances |
| Equality narrowing | `a === b` | Narrows based on literal comparisons |
| `in` operator guard | `'property' in object` | Narrows object types based on property existence |
| Truthiness narrowing | `if (value)` | Narrows away `null`, `undefined`, `0`, `""` |

**Example:**
```typescript
// 1. typeof and Truthiness narrowing
function formatValue(val: string | number | null | undefined): string {
  if (!val) {
    return "N/A"; // Narrows away null & undefined
  }
  
  if (typeof val === "number") {
    return val.toFixed(2); // Inferred as: number
  }
  
  return val.trim().toUpperCase(); // Inferred as: string
}

// 2. instanceof narrowing
function processDate(input: string | Date): number {
  if (input instanceof Date) {
    return input.getTime(); // Inferred as: Date
  }
  return new Date(input).getTime(); // Inferred as: string
}

// 3. 'in' operator narrowing
interface Admin {
  name: string;
  privileges: string[];
}

interface Customer {
  name: string;
  ordersCount: number;
}

function handleUser(user: Admin | Customer) {
  if ("privileges" in user) {
    console.log(`Admin privileges: ${user.privileges.join(", ")}`);
  } else {
    console.log(`Customer orders: ${user.ordersCount}`);
  }
}
```
