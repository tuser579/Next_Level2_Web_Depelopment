# 27. REACT & TYPESCRIPT PATTERNS

| Pattern / Hook | Type Annotation | Description |
| --- | --- | --- |
| Props Interface | `interface ButtonProps` | Strongly typed component props with optional handlers |
| Children Typing | `children: React.ReactNode` | Accepts JSX elements, strings, fragments, numbers |
| `useState<T>` | `useState<User | null>(null)` | Explicit typing for state with union/initial null |
| `useRef<T>` | `useRef<HTMLInputElement>(null)` | DOM element ref typing |
| `CSSProperties` | `style: React.CSSProperties` | Type-safe inline CSS style objects |

**Example:**
```typescript
import React, { useState, useRef } from "react";

// 1. Component Props Interface
interface ButtonProps {
  label: string;
  variant?: "primary" | "secondary" | "danger";
  disabled?: boolean;
  onClick: (e: React.MouseEvent<HTMLButtonElement>) => void;
  children?: React.ReactNode;
}

export const ActionButton: React.FC<ButtonProps> = ({
  label,
  variant = "primary",
  disabled = false,
  onClick,
  children
}) => {
  return (
    <button className={`btn btn-${variant}`} disabled={disabled} onClick={onClick}>
      {label}
      {children}
    </button>
  );
};

// 2. Typed Hooks Example
interface User {
  id: string;
  name: string;
}

export function UserProfileComponent() {
  const [user, setUser] = useState<User | null>(null);
  const inputRef = useRef<HTMLInputElement>(null);

  const handleFocus = () => {
    inputRef.current?.focus();
  };

  return (
    <div>
      <input ref={inputRef} placeholder="Enter name" />
      <button onClick={handleFocus}>Focus Input</button>
    </div>
  );
}
```
