# 26. DOM & BROWSER TYPES IN TYPESCRIPT

| DOM Type | Category | Description |
| --- | --- | --- |
| `HTMLElement` | Element Base | General HTML DOM element |
| `HTMLInputElement` | Specific Element | Input field element with `.value`, `.checked`, `.focus()` |
| `HTMLButtonElement` | Specific Element | Button element with `.disabled`, `.type` |
| `HTMLFormElement` | Specific Element | Form element with `.reset()`, `.elements` |
| `MouseEvent` | Event | Mouse click, mouse move, hover events |
| `KeyboardEvent` | Event | Key down/up events with `.key`, `.code` |
| `SubmitEvent` | Event | Form submit event with `.preventDefault()` |

**Example:**
```typescript
// 1. Typed Event Listeners & Casting
const loginForm = document.querySelector<HTMLFormElement>("#login-form");
const emailInput = document.querySelector<HTMLInputElement>("#email");

loginForm?.addEventListener("submit", (e: SubmitEvent) => {
  e.preventDefault();
  
  if (emailInput) {
    console.log("Email submitted:", emailInput.value);
  }
});

// 2. Handling Keyboard Events
window.addEventListener("keydown", (e: KeyboardEvent) => {
  if (e.key === "Escape") {
    console.log("Escape key pressed - closing modal");
  }
});

// 3. Generic querySelector utility
function getElement<T extends HTMLElement>(selector: string): T {
  const el = document.querySelector<T>(selector);
  if (!el) throw new Error(`Element with selector "${selector}" not found`);
  return el;
}

const actionBtn = getElement<HTMLButtonElement>("#action-btn");
actionBtn.disabled = true;
```
