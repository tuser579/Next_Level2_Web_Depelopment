# 8. GENERICS FUNDAMENTALS

| Generic Form | Syntax Example | Description |
| --- | --- | --- |
| Generic Function | `function identity<T>(arg: T): T` | Captures and preserves argument type throughout function execution |
| Generic Arrow Function | `const wrap = <T>(val: T) => ({ val })` | Arrow function generic syntax |
| Generic Interface | `interface ApiResponse<T> { data: T }` | Reusable interface parameterized by payload type |
| Generic Type Alias | `type Result<T, E> = ...` | Type alias with multiple type parameters |
| Generic Default Type | `interface Store<T = string>` | Fallback type if no type argument is provided |

**Example:**
```typescript
// 1. Generic Function (Preserves exact type)
function wrapInArray<T>(item: T): T[] {
  return [item];
}

const stringArray = wrapInArray("hello"); // Inferred as: string[]
const numberArray = wrapInArray(42);      // Inferred as: number[]

// 2. Generic API Response Wrapper
interface ApiResponse<TData = unknown> {
  status: "success" | "error";
  statusCode: number;
  data: TData;
  timestamp: string;
}

interface User {
  id: number;
  name: string;
}

const userResponse: ApiResponse<User> = {
  status: "success",
  statusCode: 200,
  data: { id: 1, name: "Alice" },
  timestamp: new Date().toISOString()
};

// 3. Generic Queue Class
class Queue<T> {
  private items: T[] = [];
  
  enqueue(item: T): void {
    this.items.push(item);
  }
  
  dequeue(): T | undefined {
    return this.items.shift();
  }
}

const messageQueue = new Queue<string>();
messageQueue.enqueue("Job #1");
console.log(messageQueue.dequeue()); // "Job #1"
```
