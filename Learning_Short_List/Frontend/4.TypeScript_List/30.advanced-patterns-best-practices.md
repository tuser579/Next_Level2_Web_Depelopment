# 30. ADVANCED PATTERNS & BEST PRACTICES

| Pattern / Rule | Type Technique | Description |
| --- | --- | --- |
| Branded Types (Nominal Typing) | `type UserId = string & { readonly __brand: unique symbol }` | Prevents accidentally passing raw strings where validated IDs are required |
| Exhaustiveness Checking | `const _exhaustive: never = val` | Ensures compiler flags errors when new union branches are added |
| Avoid `any` (Prefer `unknown`) | Code Quality | Maintains type safety at system boundaries |
| Builder Pattern with Generics | Fluency & Safety | Type-safe builder accumulating required fields across chained calls |

**Example:**
```typescript
// 1. Branded (Nominal) Types Pattern
// Prevents mixing up Email string with arbitrary strings or UserIDs!
type Brand<K, T> = K & { readonly __brand: T };

type EmailAddress = Brand<string, "EmailAddress">;
type UserId = Brand<string, "UserId">;

function validateEmail(raw: string): EmailAddress {
  if (!raw.includes("@")) throw new Error("Invalid email");
  return raw as EmailAddress;
}

function sendWelcomeEmail(to: EmailAddress) {
  console.log(`Sending email to: ${to}`);
}

const userEmail = validateEmail("contact@developer.com");
sendWelcomeEmail(userEmail); // ✅ Compiles cleanly!

// const plainText = "not_validated_string";
// sendWelcomeEmail(plainText); // ❌ Error: Type 'string' is not assignable to 'EmailAddress'

// 2. Exhaustive switch check with never
type PaymentMethod = "credit_card" | "paypal" | "crypto";

function processPayment(method: PaymentMethod) {
  switch (method) {
    case "credit_card": return "Charging card...";
    case "paypal": return "Redirecting to PayPal...";
    case "crypto": return "Broadcasting transaction...";
    default: {
      const _unhandled: never = method;
      throw new Error(`Unhandled method: ${_unhandled}`);
    }
  }
}
```
