# 11. USER-DEFINED TYPE GUARDS & ASSERTIONS

| Syntax / Pattern | Keyword | Description |
| --- | --- | --- |
| Type Predicate | `value is Type` | Function returning boolean that informs compiler of narrowed type on `true` |
| Assertion Function | `asserts condition` | Throws error if false; informs compiler condition holds true thereafter |
| Assertion Type Guard | `asserts value is Type` | Throws error if value doesn't match; narrows value type afterwards |

**Example:**
```typescript
// 1. User-Defined Type Guard (value is Type)
interface Fish {
  swim(): void;
}

interface Bird {
  fly(): void;
}

function isFish(pet: Fish | Bird): pet is Fish {
  return (pet as Fish).swim !== undefined;
}

function movePet(pet: Fish | Bird) {
  if (isFish(pet)) {
    pet.swim(); // Inferred as: Fish
  } else {
    pet.fly();  // Inferred as: Bird
  }
}

// 2. Assertion Function (asserts value is Type)
function assertIsString(val: unknown): asserts val is string {
  if (typeof val !== "string") {
    throw new TypeError(`Expected string, but received ${typeof val}`);
  }
}

function processInput(input: unknown) {
  assertIsString(input);
  // Beyond this line, input is strictly narrowed to string!
  console.log(input.toUpperCase());
}
```
