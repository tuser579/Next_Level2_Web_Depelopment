# 28. ASYNC TYPES & CATCH CLAUSE NARROWING

| Pattern | Safety Rule | Description |
| --- | --- | --- |
| `Promise<T>` | Typed Return | Explicitly declares resolved value type of async function |
| `catch (err: unknown)` | Type-Safe Errors | Under strict TS, caught errors are of type `unknown` and require narrowing |
| Result Pattern | `type Result<T, E>` | Functional error handling returning `{ ok: true, data } | { ok: false, error }` |

**Example:**
```typescript
// 1. Typed Async Function
interface Post {
  id: number;
  title: string;
}

async function fetchPost(postId: number): Promise<Post> {
  const res = await fetch(`https://jsonplaceholder.typicode.com/posts/${postId}`);
  if (!res.ok) {
    throw new Error(`HTTP Error ${res.status}`);
  }
  return (await res.json()) as Post;
}

// 2. Type-Safe Catch Block (Narrowing unknown error)
async function executeTask() {
  try {
    const post = await fetchPost(1);
    console.log(post.title);
  } catch (err: unknown) {
    if (err instanceof Error) {
      console.error(`Error message: ${err.message}`);
    } else {
      console.error("Unknown error occurred:", err);
    }
  }
}

// 3. Functional Result Type Pattern (No throw)
type Result<T, E = Error> = 
  | { success: true; data: T }
  | { success: false; error: E };

async function safeFetchUser(id: number): Promise<Result<Post>> {
  try {
    const data = await fetchPost(id);
    return { success: true, data };
  } catch (err) {
    return { 
      success: false, 
      error: err instanceof Error ? err : new Error(String(err)) 
    };
  }
}
```
