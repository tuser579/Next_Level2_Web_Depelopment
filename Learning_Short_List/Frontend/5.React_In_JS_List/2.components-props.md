# 2. COMPONENTS, PROPS & CHILDREN

| Concept | Pattern / Syntax | Description |
| --- | --- | --- |
| Functional Component | `function Card(props) { return ...; }` | JavaScript function returning JSX UI representation |
| Props Destructuring | `function Button({ label, onClick, variant = 'primary' })` | Unpacks properties with default values directly in parameter signature |
| Readonly Props | `props.title = 'New'; // ❌ ERROR` | Props are strictly immutable (pure input flow from parent to child) |
| `children` Prop | `<Card><p>Content</p></Card>` | Passes nested elements and components into component body |
| Function Props (Callbacks) | `<SearchInput onSearch={handleSearch} />` | Passes handler functions to child to send data back up to parent |

**Example:**
```jsx
import React from "react";

// Child Component with Destructured Props & Default Values
function ActionButton({ label, onClick, variant = "primary", disabled = false }) {
  return (
    <button 
      className={`btn btn-${variant}`} 
      onClick={onClick} 
      disabled={disabled}
    >
      {label}
    </button>
  );
}

// Wrapper Component demonstrating 'children' prop
function CardWrapper({ title, children }) {
  return (
    <section className="card-container">
      <header className="card-header">
        <h3>{title}</h3>
      </header>
      <div className="card-body">
        {children} {/* Renders whatever is passed inside the tags */}
      </div>
    </section>
  );
}

// Parent Component
export default function App() {
  const handleClick = (actionName) => {
    alert(`Triggered action: ${actionName}`);
  };

  return (
    <CardWrapper title="User Management">
      <p>Manage permissions and access tokens.</p>
      <ActionButton 
        label="Delete Account" 
        variant="danger" 
        onClick={() => handleClick("Delete")} 
      />
    </CardWrapper>
  );
}
```
