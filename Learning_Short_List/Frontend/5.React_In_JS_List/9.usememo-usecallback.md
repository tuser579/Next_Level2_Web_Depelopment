# 9. USEMEMO & USECALLBACK HOOKS

| Hook | Returns | Primary Purpose |
| --- | --- | --- |
| `useMemo(fn, deps)` | Memoized computed VALUE | Avoids recalculating expensive, CPU-heavy computations on unrelated re-renders |
| `useCallback(fn, deps)` | Memoized FUNCTION reference | Prevents recreation of callback functions passed as props to memoized child components |
| When to use | Performance Optimization | Only when benchmarking confirms expensive operations or child re-renders are problematic |

**Example:**
```jsx
import React, { useState, useMemo, useCallback } from "react";

// Memoized Child Component
const FilterButton = React.memo(({ onFilter }) => {
  console.log("FilterButton re-rendered!");
  return <button onClick={onFilter}>Apply Filter</button>;
});

export default function DataDashboard() {
  const [numbers] = useState([5, 12, 8, 130, 44, 99, 105]);
  const [darkMode, setDarkMode] = useState(false);
  const [filterThreshold, setFilterThreshold] = useState(10);

  // 1. useMemo: Caches heavy calculation result unless dependencies change
  const totalSum = useMemo(() => {
    console.log("Computing sum...");
    return numbers.reduce((sum, n) => sum + n, 0);
  }, [numbers]);

  // 2. useCallback: Caches function reference so FilterButton doesn't re-render on darkMode toggle!
  const handleFilter = useCallback(() => {
    console.log(`Filtering numbers greater than ${filterThreshold}`);
  }, [filterThreshold]);

  return (
    <div style={{ background: darkMode ? "#333" : "#fff", color: darkMode ? "#fff" : "#000", padding: 20 }}>
      <h3>Total Sum: {totalSum}</h3>
      <FilterButton onFilter={handleFilter} />
      
      <button onClick={() => setDarkMode(prev => !prev)}>
        Toggle Theme (Does NOT re-render FilterButton!)
      </button>
    </div>
  );
}
```
