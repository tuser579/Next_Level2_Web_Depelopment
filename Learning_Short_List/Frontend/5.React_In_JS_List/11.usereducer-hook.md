# 11. THE USEREDUCER HOOK

| Concept | Role | Description |
| --- | --- | --- |
| `useReducer(reducer, initialArg)` | Hook | Alternative to `useState` for complex state logic involving multiple sub-values |
| `state` | Current State | The read-only immutable state tree |
| `dispatch(action)` | Trigger | Dispatches an action object `{ type, payload }` to trigger state transitions |
| `reducer(state, action)` | Pure Function | Takes current state and action, returns NEW state without mutation |

**Example:**
```jsx
import React, { useReducer } from "react";

// 1. Initial State
const initialState = {
  cart: [],
  total: 0
};

// 2. Pure Reducer Function
function cartReducer(state, action) {
  switch (action.type) {
    case "ADD_ITEM": {
      const updatedCart = [...state.cart, action.payload];
      return {
        cart: updatedCart,
        total: state.total + action.payload.price
      };
    }
    case "REMOVE_ITEM": {
      const updatedCart = state.cart.filter(item => item.id !== action.payload.id);
      return {
        cart: updatedCart,
        total: state.total - action.payload.price
      };
    }
    case "CLEAR_CART":
      return initialState;
    default:
      throw new Error(`Unknown action type: ${action.type}`);
  }
}

// 3. Component
export default function ShoppingCart() {
  const [state, dispatch] = useReducer(cartReducer, initialState);

  const addItem = () => {
    const newItem = { id: Date.now(), name: "Book", price: 15 };
    dispatch({ type: "ADD_ITEM", payload: newItem });
  };

  return (
    <div>
      <h3>Total Cart Items: {state.cart.length} | Total Cost: ${state.total}</h3>
      <button onClick={addItem}>Add $15 Book</button>
      <button onClick={() => dispatch({ type: "CLEAR_CART" })}>Clear Cart</button>

      <ul>
        {state.cart.map(item => (
          <li key={item.id}>
            {item.name} (${item.price})
            <button onClick={() => dispatch({ type: "REMOVE_ITEM", payload: item })}>
              Remove
            </button>
          </li>
        ))}
      </ul>
    </div>
  );
}
```
