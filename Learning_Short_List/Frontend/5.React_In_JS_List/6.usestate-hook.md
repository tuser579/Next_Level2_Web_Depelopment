# 6. THE USESTATE HOOK (PRIMITIVES & OBJECTS)

| Pattern | Syntax Example | Description |
| --- | --- | --- |
| Basic State | `const [count, setCount] = useState(0);` | Initializes state with default value |
| Functional Updates | `setCount(prev => prev + 1)` | Guarantees latest state value during batch updates |
| Object State (Spread) | `setUser(prev => ({ ...prev, name: 'New' }))` | State does NOT auto-merge; must spread previous properties |
| Array State (Immutability) | `setItems(prev => [...prev, newItem])` | Add, update, and remove array items immutably |
| Lazy Initial State | `useState(() => computeExpensiveInitialState())` | Passes function to run expensive initializer ONLY on first mount |

**Example:**
```jsx
import React, { useState } from "react";

export default function UserProfileManager() {
  // 1. Primitive State with Functional Updates
  const [likes, setLikes] = useState(0);

  const handleDoubleLike = () => {
    // Functional updater ensures both increments execute correctly!
    setLikes(prev => prev + 1);
    setLikes(prev => prev + 1);
  };

  // 2. Object State (Maintaining Immutability)
  const [profile, setProfile] = useState({
    username: "coder_99",
    email: "coder@example.com",
    theme: "dark"
  });

  const updateEmail = (newEmail) => {
    setProfile(prev => ({
      ...prev,
      email: newEmail // Overwrites only email
    }));
  };

  // 3. Array State (Immutable additions and deletions)
  const [skills, setSkills] = useState(["React", "JavaScript"]);

  const addSkill = (skill) => {
    setSkills(prev => [...prev, skill]);
  };

  const removeSkill = (indexToRemove) => {
    setSkills(prev => prev.filter((_, idx) => idx !== indexToRemove));
  };

  return (
    <div className="profile-box">
      <h2>Profile: {profile.username}</h2>
      <p>Email: {profile.email}</p>
      <button onClick={() => updateEmail("new_email@test.com")}>Change Email</button>
      
      <p>Likes: {likes}</p>
      <button onClick={handleDoubleLike}>Double Like (+2)</button>

      <h4>Skills:</h4>
      <ul>
        {skills.map((skill, idx) => (
          <li key={idx}>
            {skill} <button onClick={() => removeSkill(idx)}>❌</button>
          </li>
        ))}
      </ul>
      <button onClick={() => addSkill("TypeScript")}>Add TypeScript</button>
    </div>
  );
}
```
