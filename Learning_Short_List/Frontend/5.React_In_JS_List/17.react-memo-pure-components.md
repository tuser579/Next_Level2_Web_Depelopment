# 17. REACT.MEMO & PURE COMPONENTS

| Tool / Technique | Comparison Mode | Description |
| --- | --- | --- |
| `React.memo(Component)` | Shallow comparison | Higher-Order Component that skips re-rendering if props are shallowly identical |
| Custom `arePropsEqual` | `React.memo(Comp, (prev, next) => ...)` | Custom function returning `true` to skip re-render |
| Pairing with `useCallback` | Function props | Required to prevent new function references from busting `React.memo` |

**Example:**
```jsx
import React, { useState, useCallback } from "react";

// Pure Memoized Component
const ExpensiveUserCard = React.memo(({ user, onSelect }) => {
  console.log(`Rendering ExpensiveUserCard for ${user.name}`);
  return (
    <div className="user-card" onClick={() => onSelect(user.id)}>
      <h4>{user.name}</h4>
      <p>{user.email}</p>
    </div>
  );
});

export default function UserList() {
  const [users] = useState([
    { id: 1, name: "Alice", email: "alice@test.com" },
    { id: 2, name: "Bob", email: "bob@test.com" }
  ]);
  const [counter, setCounter] = useState(0);

  // useCallback ensures onSelect function reference stays stable
  const handleSelect = useCallback((id) => {
    console.log("Selected user ID:", id);
  }, []);

  return (
    <div>
      <p>Unrelated Counter: {counter}</p>
      <button onClick={() => setCounter(c => c + 1)}>Increment (Won't re-render cards!)</button>

      {users.map(u => (
        <ExpensiveUserCard key={u.id} user={u} onSelect={handleSelect} />
      ))}
    </div>
  );
}
```
