# 7. THE USEEFFECT HOOK & LIFECYCLE

| Dependency Array | Trigger Condition | Simulated Lifecycle |
| --- | --- | --- |
| No Dependencies (`useEffect(fn)`) | Runs after EVERY render | Component Did Mount & Component Did Update |
| Empty Array (`useEffect(fn, [])`) | Runs ONCE on initial mount | Component Did Mount |
| With Dependencies (`useEffect(fn, [id])`) | Runs when `id` changes | Component Did Update for specific value |
| Cleanup Function (`return () => {...}`) | Runs before re-running effect or before component unmounts | Component Will Unmount & Sub-cleanup |

**Example:**
```jsx
import React, { useState, useEffect } from "react";

export default function RealTimeClock() {
  const [time, setTime] = useState(new Date().toLocaleTimeString());
  const [userId, setUserId] = useState(1);
  const [userData, setUserData] = useState(null);

  // 1. Timer with Cleanup (Unmount cleanup)
  useEffect(() => {
    const timerId = setInterval(() => {
      setTime(new Date().toLocaleTimeString());
    }, 1000);

    // Cleanup function: Prevents memory leaks when component unmounts!
    return () => {
      clearInterval(timerId);
    };
  }, []); // Empty array = mount only

  // 2. Fetch Data when dependency changes
  useEffect(() => {
    let isMounted = true;

    async function fetchUser() {
      const res = await fetch(`https://jsonplaceholder.typicode.com/users/${userId}`);
      const data = await res.json();
      if (isMounted) setUserData(data);
    }

    fetchUser();

    // Cleanup cancels updating unmounted component
    return () => {
      isMounted = false;
    };
  }, [userId]); // Re-runs whenever userId changes

  return (
    <div>
      <h3>Current Time: {time}</h3>
      <button onClick={() => setUserId(prev => prev + 1)}>Next User</button>
      {userData ? <p>User: {userData.name}</p> : <p>Loading user...</p>}
    </div>
  );
}
```
