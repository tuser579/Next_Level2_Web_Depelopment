# 13. FORMS & CONTROLLED COMPONENTS

| Element | Binding Property | Change Event Handler |
| --- | --- | --- |
| Text Input / Textarea | `value={state}` | `onChange={e => setState(e.target.value)}` |
| Checkbox | `checked={state}` | `onChange={e => setState(e.target.checked)}` |
| Select Dropdown | `value={state}` | `onChange={e => setState(e.target.value)}` |
| Multi-Input Pattern | `[e.target.name]: value` | Single handler managing entire form object state |

**Example:**
```jsx
import React, { useState } from "react";

export default function RegistrationForm() {
  const [formData, setFormData] = useState({
    username: "",
    email: "",
    country: "US",
    agreeToTerms: false
  });

  // Single dynamic change handler for all form fields
  const handleChange = (e) => {
    const { name, value, type, checked } = e.target;
    setFormData(prev => ({
      ...prev,
      [name]: type === "checkbox" ? checked : value
    }));
  };

  const handleSubmit = (e) => {
    e.preventDefault();
    console.log("Submitting form:", formData);
  };

  return (
    <form onSubmit={handleSubmit} className="form-box">
      <div>
        <label>Username:</label>
        <input 
          name="username" 
          value={formData.username} 
          onChange={handleChange} 
          required 
        />
      </div>

      <div>
        <label>Email:</label>
        <input 
          name="email" 
          type="email" 
          value={formData.email} 
          onChange={handleChange} 
          required 
        />
      </div>

      <div>
        <label>Country:</label>
        <select name="country" value={formData.country} onChange={handleChange}>
          <option value="US">United States</option>
          <option value="CA">Canada</option>
          <option value="UK">United Kingdom</option>
        </select>
      </div>

      <div>
        <label>
          <input 
            type="checkbox" 
            name="agreeToTerms" 
            checked={formData.agreeToTerms} 
            onChange={handleChange} 
          />
          I agree to the terms
        </label>
      </div>

      <button type="submit" disabled={!formData.agreeToTerms}>Register</button>
    </form>
  );
}
```
