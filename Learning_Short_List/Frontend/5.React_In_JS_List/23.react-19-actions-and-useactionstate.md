# 23. REACT 19: ACTIONS, USEACTIONSTATE & USEOPTIMISTIC

| React 19 Feature | Hook / Prop | Description |
| --- | --- | --- |
| Form Actions | `<form action={asyncAction}>` | Passes async functions directly into form `action` prop |
| `useActionState()` | `[state, formAction, isPending] = useActionState(...)` | Manages form state, validation errors, and pending status automatically |
| `useFormStatus()` | `const { pending } = useFormStatus()` | Reads parent `<form>` pending submission state inside child components |
| `useOptimistic()` | `const [optimistic, setOptimistic] = useOptimistic(...)` | Renders instant optimistic UI updates while async action is in flight |

**Example:**
```jsx
import React, { useActionState, useOptimistic } from "react";

// Async Server/Client Action
async function updateUsernameAction(previousState, formData) {
  const newName = formData.get("username");
  await new Promise(r => setTimeout(r, 1000)); // Simulate API delay
  
  if (!newName) return { error: "Username cannot be empty", name: previousState.name };
  return { error: null, name: newName };
}

export default function ProfileEditor() {
  // useActionState handles state, action dispatcher, and isPending automatically
  const [state, formAction, isPending] = useActionState(updateUsernameAction, {
    name: "Alex",
    error: null
  });

  // Optimistic UI updates
  const [optimisticName, setOptimisticName] = useOptimistic(
    state.name,
    (current, update) => update
  );

  return (
    <form action={async (formData) => {
      setOptimisticName(formData.get("username"));
      await formAction(formData);
    }}>
      <p>Current Name: <strong>{optimisticName}</strong> {isPending && "⏳ Updating..."}</p>
      <input name="username" defaultValue={state.name} />
      <button type="submit" disabled={isPending}>Save</button>
      {state.error && <p style={{ color: "red" }}>{state.error}</p>}
    </form>
  );
}
```
