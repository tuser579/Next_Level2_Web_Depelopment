# 8. THE USEREF HOOK

| Use Case | Syntax Example | Description |
| --- | --- | --- |
| Direct DOM Access | `const inputRef = useRef(null);` `<input ref={inputRef} />` | Accesses raw browser DOM element (e.g. `.focus()`, `.scrollIntoView()`) |
| Mutable Value (No Re-render) | `const timerRef = useRef(null);` | Stores mutable values that persist across renders without triggering a re-render |
| Previous State Tracking | `const prevCount = useRef();` | Captures previous render's value in `useEffect` |

**Example:**
```jsx
import React, { useState, useRef, useEffect } from "react";

export default function AutoFocusAndTimer() {
  const [seconds, setSeconds] = useState(0);
  const inputRef = useRef(null);     // 1. DOM Element Reference
  const timerIdRef = useRef(null);   // 2. Mutable value across renders
  const prevSeconds = useRef(0);     // 3. Previous value tracking

  // Auto-focus input on initial mount
  useEffect(() => {
    inputRef.current?.focus();
  }, []);

  // Track previous value
  useEffect(() => {
    prevSeconds.current = seconds;
  }, [seconds]);

  const startTimer = () => {
    if (timerIdRef.current !== null) return;
    timerIdRef.current = setInterval(() => {
      setSeconds(prev => prev + 1);
    }, 1000);
  };

  const stopTimer = () => {
    clearInterval(timerIdRef.current);
    timerIdRef.current = null; // Reset ref
  };

  return (
    <div className="card">
      <input ref={inputRef} type="text" placeholder="Auto-focused input..." />
      <hr />
      <p>Current Seconds: {seconds}</p>
      <p>Previous Seconds: {prevSeconds.current}</p>
      <button onClick={startTimer}>Start Timer</button>
      <button onClick={stopTimer}>Stop Timer</button>
    </div>
  );
}
```
