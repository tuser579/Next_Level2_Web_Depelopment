# 27. ROUTING: REACT ROUTER (V6 & V7)

| Component / Hook | Syntax / Role | Description |
| --- | --- | --- |
| `createBrowserRouter` | `const router = createBrowserRouter([...])` | Data-router configuration with nested route layouts |
| `<RouterProvider>` | `<RouterProvider router={router} />` | Root provider providing routing context |
| `<Outlet />` | Layout placeholder | Renders matching child route components inside parent layout |
| `useParams()` | `const { id } = useParams()` | Extracts dynamic URL parameters (`/users/:id`) |
| `useNavigate()` | `const navigate = useNavigate()` | Programmatic navigation (`navigate('/dashboard')`) |

**Example:**
```jsx
import React from "react";
import { 
  createBrowserRouter, 
  RouterProvider, 
  Outlet, 
  Link, 
  useParams, 
  useNavigate 
} from "react-router-dom";

// Layout Component
function RootLayout() {
  return (
    <div>
      <nav style={{ display: "flex", gap: 15, padding: 10, background: "#eee" }}>
        <Link to="/">Home</Link>
        <Link to="/users/42">User #42</Link>
      </nav>
      <main style={{ padding: 20 }}>
        <Outlet /> {/* Child routes render here */}
      </main>
    </div>
  );
}

// User Detail Page
function UserDetail() {
  const { id } = useParams();
  const navigate = useNavigate();

  return (
    <div>
      <h2>User Profile ID: {id}</h2>
      <button onClick={() => navigate("/")}>Go Home</button>
    </div>
  );
}

// Router Configuration
const router = createBrowserRouter([
  {
    path: "/",
    element: <RootLayout />,
    children: [
      { index: true, element: <h3>Home Dashboard</h3> },
      { path: "users/:id", element: <UserDetail /> }
    ]
  }
]);

export default function App() {
  return <RouterProvider router={router} />;
}
```
