# 12. CREATING CUSTOM HOOKS

| Rule / Feature | Requirement | Description |
| --- | --- | --- |
| Naming Convention | Must start with `use` | e.g. `useFetch`, `useLocalStorage`, `useWindowSize` |
| State Encapsulation | Independent state per call | Each component calling custom hook gets its own isolated instance of state |
| Reusability | Composable logic | Extracts repetitive effect and state handling across components |

**Example:**
```jsx
import { useState, useEffect } from "react";

// 1. Custom Hook for LocalStorage synchronization
export function useLocalStorage(key, initialValue) {
  const [storedValue, setStoredValue] = useState(() => {
    try {
      const item = localStorage.getItem(key);
      return item ? JSON.parse(item) : initialValue;
    } catch (error) {
      return initialValue;
    }
  });

  useEffect(() => {
    try {
      localStorage.setItem(key, JSON.stringify(storedValue));
    } catch (error) {
      console.error(error);
    }
  }, [key, storedValue]);

  return [storedValue, setStoredValue];
}

// 2. Custom Hook for Window Resize Detection
export function useWindowSize() {
  const [size, setSize] = useState({
    width: window.innerWidth,
    height: window.innerHeight
  });

  useEffect(() => {
    const handleResize = () => setSize({ width: window.innerWidth, height: window.innerHeight });
    window.addEventListener("resize", handleResize);
    return () => window.removeEventListener("resize", handleResize);
  }, []);

  return size;
}

// 3. Usage in component
export default function Preferences() {
  const [theme, setTheme] = useLocalStorage("app_theme", "light");
  const { width } = useWindowSize();

  return (
    <div>
      <p>Theme from storage: {theme}</p>
      <p>Window width: {width}px</p>
      <button onClick={() => setTheme(theme === "light" ? "dark" : "light")}>
        Toggle Saved Theme
      </button>
    </div>
  );
}
```
