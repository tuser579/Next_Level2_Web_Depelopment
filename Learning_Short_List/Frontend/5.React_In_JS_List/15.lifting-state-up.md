# 15. LIFTING STATE UP

| Concept | Pattern | Description |
| --- | --- | --- |
| Common Parent | Ancestor Component | Holds single source of truth state shared between multiple sibling components |
| Downwards Flow | Props | State passed down to child components as read-only props |
| Upwards Flow | Callback Handlers | Children call parent callback functions to request state updates |

**Example:**
```jsx
import React, { useState } from "react";

// Sibling 1: Input control
function TemperatureInput({ scale, temperature, onTemperatureChange }) {
  return (
    <fieldset>
      <legend>Enter temperature in {scale === "c" ? "Celsius" : "Fahrenheit"}:</legend>
      <input 
        value={temperature} 
        onChange={(e) => onTemperatureChange(e.target.value)} 
      />
    </fieldset>
  );
}

// Sibling 2: Boiling evaluator
function BoilingVerdict({ celsius }) {
  if (parseFloat(celsius) >= 100) {
    return <p style={{ color: "red" }}>🔥 The water would boil.</p>;
  }
  return <p style={{ color: "blue" }}>💧 The water would not boil.</p>;
}

// Parent: Holds the single source of truth state
export default function Calculator() {
  const [temperature, setTemperature] = useState("");

  return (
    <div>
      <h3>Temperature Converter</h3>
      <TemperatureInput 
        scale="c" 
        temperature={temperature} 
        onTemperatureChange={setTemperature} 
      />
      <BoilingVerdict celsius={temperature} />
    </div>
  );
}
```
