# 14. UNCONTROLLED COMPONENTS & FORMDATA

| Technique | Mechanism | Best Used For |
| --- | --- | --- |
| Uncontrolled Inputs via `useRef` | `ref={fileInputRef}` | File uploads, non-React integration (e.g. jQuery/Canvas plugins) |
| Native `FormData` in `onSubmit` | `new FormData(e.currentTarget)` | Standard forms where intermediate character-by-character state is unnecessary |
| `defaultValue` / `defaultChecked` | `defaultValue="John"` | Sets initial uncontrolled value without forcing controlled component lifecycle |

**Example:**
```jsx
import React, { useRef } from "react";

export default function FileUploadForm() {
  const fileInputRef = useRef(null);

  const handleSubmit = (e) => {
    e.preventDefault();
    
    // Extract form data natively from form element
    const formData = new FormData(e.currentTarget);
    const title = formData.get("docTitle");
    const file = fileInputRef.current.files[0];

    console.log("Title:", title);
    console.log("File attached:", file?.name, file?.size, "bytes");
  };

  return (
    <form onSubmit={handleSubmit}>
      <div>
        <label>Document Title:</label>
        {/* defaultValue sets initial value for uncontrolled input */}
        <input name="docTitle" defaultValue="Report 2026" />
      </div>

      <div>
        <label>Select File (Always Uncontrolled):</label>
        <input ref={fileInputRef} type="file" required />
      </div>

      <button type="submit">Upload Document</button>
    </form>
  );
}
```
