# 21. CONCURRENCY: USETRANSITION & USEDEFERREDVALUE

| Hook | Returns | Description |
| --- | --- | --- |
| `useTransition()` | `[isPending, startTransition]` | Marks state update as non-urgent / transition so UI remains responsive to urgent user typing |
| `useDeferredValue(value)` | Deferred copy of value | Defers updating heavy calculation until urgent render passes finish |

**Example:**
```jsx
import React, { useState, useTransition, useDeferredValue } from "react";

export default function SearchPage() {
  const [input, setInput] = useState("");
  const [isPending, startTransition] = useTransition();
  const [searchQuery, setSearchQuery] = useState("");

  const handleChange = (e) => {
    // Urgent update: Updates input field immediately!
    setInput(e.target.value);

    // Non-urgent transition update: Renders heavy filtered list in background
    startTransition(() => {
      setSearchQuery(e.target.value);
    });
  };

  return (
    <div>
      <input value={input} onChange={handleChange} placeholder="Search 10,000 items..." />
      {isPending && <p>⏳ Updating search results...</p>}
      <HeavyList query={searchQuery} />
    </div>
  );
}

function HeavyList({ query }) {
  // Alternatively defer value
  const deferredQuery = useDeferredValue(query);
  const items = Array.from({ length: 5000 }, (_, i) => `Item ${i} matches ${deferredQuery}`);

  return (
    <ul>
      {items.slice(0, 50).map((item, idx) => <li key={idx}>{item}</li>)}
    </ul>
  );
}
```
